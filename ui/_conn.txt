component ConnectButton inherits Rectangle {
    in property <ConnState> state;
    callback clicked();
    width: 190px;
    height: 190px;
    // Форма «перетекает» из скруглённого квадрата в круг — в духе M3 Expressive.
    border-radius: root.state == ConnState.on ? 95px : 64px;
    background: root.state == ConnState.on ? Theme.primary
        : root.state == ConnState.connecting ? Theme.primary-container
        : Theme.container-highest;
    animate border-radius { duration: 350ms; easing: ease-in-out; }
    animate background { duration: 250ms; easing: ease-in-out; }
    Rectangle {
        border-radius: parent.border-radius;
        background: (root.state == ConnState.on ? Theme.on-primary : Theme.on-surface).with-alpha(ta.pressed ? 0.12 : ta.has-hover ? 0.08 : 0);
    }
    Path {
        x: (parent.width - self.width) / 2;
        y: (parent.height - self.height) / 2;
        width: 72px;
        height: 72px;
        viewbox-width: 24;
        viewbox-height: 24;
        commands: "M12 3v9 M6.4 6.8a8 8 0 1 0 11.2 0";
        stroke: root.state == ConnState.on ? Theme.on-primary
            : root.state == ConnState.connecting ? Theme.on-primary-container
            : Theme.primary;
        stroke-width: 2px;
        stroke-line-cap: round;
        stroke-line-join: round;
        opacity: root.state == ConnState.connecting ? 0.45 + 0.55 * Math.abs(Math.sin(animation-tick() / 700ms * 180deg)) : 1;
    }
    ta := TouchArea {
        enabled: root.state != ConnState.connecting;
        clicked => { root.clicked(); }
    }
}
